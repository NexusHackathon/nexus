#include <SPI.h>
#include <LoRa.h>
#include <WiFi.h>
#include <WebServer.h>

const char* ssid = "4GWIFI_66505"; // לשנות לווייפיי שלך
const char* password = "1234567890"; // לשנות לסיסמה שלך

WebServer server(8000);

// הוספנו את ה-yolo לטקסט ההתחלתי
String latestJsonData = "{\"gas\":0,\"mag\":2500,\"rssi\":0.0,\"yolo\":0,\"c_z\":0,\"c_n\":0,\"c_f\":0}";

void setup() {
  Serial.begin(9600);
  while (!Serial);

  LoRa.setPins(5, 14, 2);
  if (!LoRa.begin(868E6)) {
    Serial.println("Starting LoRa failed!");
    while (1);
  }

  Serial.print("Connecting to WiFi");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\n>>> WiFi Connected! <<<");
  Serial.print("IP Address for Python: ");
  Serial.println(WiFi.localIP());

  server.on("/data", []() {
    server.send(200, "application/json", latestJsonData);
  });

  server.begin();
}

void loop() {
  server.handleClient();

  int packetSize = LoRa.parsePacket();
  if (packetSize) {
    String incomingJson = "";
    while (LoRa.available()) {
      incomingJson += (char)LoRa.read();
    }
    latestJsonData = incomingJson;
    Serial.println("New Data: " + incomingJson);
  }
}
